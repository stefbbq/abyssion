+  childKey: string,
+  from: string,
+  to: string,

+): ActionZoneAnimationVariant | undefined => {
+  const childMap = config.childTransitions?.[childKey]

+  if (childMap[from] && childMap[from][to]) return childMap[from][to]
+  if (childMap[from] && childMap[from].default) return childMap[from].default
+  if (childMap.default && childMap.default[to]) return childMap.default[to]

+/**
+ * Recursively resolves the most specific config node for a given route, key, and selector path.
+ * Looks up in order: exact route, route-pair, selector, key, '*', '/*', and falls back to parent/default.
+ *
+ * @param config - the root config object (collapsed, collapsedPage, expanded, etc)
+ * @param route - the current route (e.g., '/shows')
+ * @param keyPath - array of keys/selectors from root to target child (e.g., ['contact', 'icon'])
+ * @returns the resolved config node (with animation/style/layout/children)
+ *
+ * @example
+ *   const node = resolveActionZoneConfigNode(collapsed, '/shows', ['contact', 'icon'])
+ *   // node.animation, node.style, node.layout, node.children
+ */
+export const resolveActionZoneConfigNode = (
+  config: any,
+  route: string,
+  keyPath: string[],
+): any => {
+  let node = config[route]
+  if (!node && route.includes('->')) node = config[route]
+  if (!node) node = config['/*']
+  if (!node) node = config['*']
+  if (!node) return undefined
+  if (!keyPath.length) return node
+  const [currentKey, ...rest] = keyPath
+  if (node.children && node.children[currentKey]) {
+    return resolveActionZoneConfigNode(node.children[currentKey], route, rest)
+  }
+  if (node.children) {
+    if (node.children['.' + currentKey]) return resolveActionZoneConfigNode(node.children['.' + currentKey], route, rest)
+    if (node.children['#' + currentKey]) return resolveActionZoneConfigNode(node.children['#' + currentKey], route, rest)
+    if (node.children['*']) return resolveActionZoneConfigNode(node.children['*'], route, rest)
+  }
+  return node
+}
